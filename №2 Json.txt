INSERT INTO customer_types (name)
SELECT DISTINCT type FROM raw_customers;
INSERT INTO order_statuses (name)
SELECT DISTINCT status FROM raw_orders;
INSERT INTO categories (name)
SELECT DISTINCT category FROM raw_products;
INSERT INTO material_types (name)
SELECT DISTINCT type FROM raw_materials;
INSERT INTO customers
  (id, name, inn, address, phone, customer_type_id)
SELECT c.id, c.name, NULLIF(c.inn, ''), c.addres, c.phone, t.id
FROM raw_customers c
JOIN customer_types t ON t.name = c.type;
INSERT INTO orders
  (id, order_date, customer_id, order_status_id)
SELECT o.id, o.order_date, o.customer_id, s.id
FROM raw_orders o
JOIN order_statuses s ON s.name = o.status;
INSERT INTO products (id, name, category_id)
SELECT p.id, p.name, c.id
FROM raw_products p
JOIN categories c ON c.name = p.category
WHERE p.n IN (SELECT MIN(n) FROM raw_products GROUP BY id);
INSERT INTO materials (id, name, material_type_id)
SELECT m.id, m.name, t.id
FROM raw_materials m
JOIN material_types t ON t.name = m.type
WHERE m.n IN (SELECT MIN(n) FROM raw_materials GROUP BY id);

INSERT INTO order_items (id, order_id, product_id, quantity)
SELECT id, order_id, product_id, quantity
FROM raw_order_items
WHERE n IN (SELECT MIN(n) FROM raw_order_items GROUP BY id)
  AND order_id IN (SELECT id FROM orders)
  AND product_id IN (SELECT id FROM products);
INSERT INTO specifications (id, product_id, material_id, qty)
SELECT id, product_id, material_id, qty
FROM raw_specification
WHERE n IN (SELECT MIN(n) FROM raw_specification GROUP BY id)
  AND product_id IN (SELECT id FROM products)
  AND material_id IN (SELECT id FROM materials);
INSERT INTO prices (id, product_id, material_id, price)
SELECT id,
       CASE WHEN item_type = 'Товар' THEN item_id END,
       CASE WHEN item_type <> 'Товар' THEN item_id END,
       price
FROM raw_prices
WHERE n IN (SELECT MIN(n) FROM raw_prices GROUP BY id)
  AND ((item_type = 'Товар'
        AND item_id IN (SELECT id FROM products))
    OR (item_type <> 'Товар'
        AND item_id IN (SELECT id FROM materials)));
INSERT INTO discounts
  (id, product_id, material_id, discount_percent)
SELECT id,
       CASE WHEN item_type = 'Товар' THEN item_id END,
       CASE WHEN item_type <> 'Товар' THEN item_id END,
       discount_percent
FROM raw_sales
WHERE n IN (SELECT MIN(n) FROM raw_sales GROUP BY id)
  AND ((item_type = 'Товар'
        AND item_id IN (SELECT id FROM products))
    OR (item_type <> 'Товар'
        AND item_id IN (SELECT id FROM materials)));
